import React from "react";
import { useLocation } from "react-router-dom";

// Props expected: userName, userId
// Date comes from navigate state from Dashboard.jsx


const DailySchedule = ({ userName, userId }) => {
const location = useLocation();
const selectedDate = location.state?.date || "No date selected";


const daysOfWeek = [
"Monday",
"Tuesday",
"Wednesday",
"Thursday",
"Friday",
"Saturday",
"Sunday",
];


// Generate 30-minute intervals from 6:00 to 22:00
const generateTimeSlots = () => {
const slots = [];
let hour = 6;
let minute = 0;


while (hour < 22 || (hour === 22 && minute === 0)) {
const formatted = `${hour.toString().padStart(2, "0")}:${minute
.toString()
.padStart(2, "0")}`;
slots.push(formatted);


minute += 30;
if (minute === 60) {
minute = 0;
hour++;
}
}
return slots;
};


const timeSlots = generateTimeSlots();
const exerciseTypes = ["Exercise A", "Exercise B", "Exercise C", "Rest", "Custom"];


return (
<div className="schedule-container" style={{ padding: "20px", fontFamily: "Arial" }}>
<h2>Daily Schedule</h2>


<div style={{ marginBottom: "20px" }}>
<strong>User Name:</strong> {userName} <br />
<strong>User ID:</strong> {userId} <br />
<strong>Date Selected:</strong> {selectedDate}
</div>


<div className="schedule-grid" style={{ display: "grid", gridTemplateColumns: "150px 1fr", gap: "10px" }}>
{daysOfWeek.map((day) => (
<div key={day} style={{ border: "1px solid #ddd", padding: "10px", borderRadius: "6px" }}>
<h3 style={{ marginTop: 0 }}>{day}</h3>
<div>
{timeSlots.map((time, index) => (
<div
key={index}
style={{
display: "flex",
justifyContent: "space-between",
padding: "6px 0",
borderBottom: "1px solid #eee",
}}
>
<span>{time}</span>
<select>
{exerciseTypes.map((ex, i) => (
<option key={i} value={ex}>
{ex}
</option>
))}
</select>
</div>
))}
</div>
</div>
))}
</div>
</div>
);
};


export default DailySchedule;
